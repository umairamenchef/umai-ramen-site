```js
import { pathToFileURL } from 'node:url';
const { chromium } = await import(pathToFileURL(`${process.env.PW_DIR}/index.mjs`).href);
const [base = 'http://localhost:3100', outDir = '.'] = process.argv.slice(2);
const browser = await chromium.launch();
const context = await browser.newContext({ viewport: { width: 375, height: 812 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true });
let failures = 0;
for (const locale of ['fr', 'en', 'de', 'it', 'zh']) {
  const page = await context.newPage();
  const response = await page.goto(`${base}/${locale}/menu`, { waitUntil: 'networkidle' });
  const r = await page.evaluate(() => {
    const pills = [...document.querySelectorAll('[data-menu-locale-switcher] a')].map((a) => {
      const b = a.getBoundingClientRect();
      return { href: a.getAttribute('href'), w: Math.round(b.width), h: Math.round(b.height), current: a.getAttribute('aria-current') };
    });
    const h2 = document.querySelector('main h2');
    return {
      lang: document.documentElement.lang,
      scrollWidth: document.documentElement.scrollWidth,
      clientWidth: document.documentElement.clientWidth,
      pills,
      stickyNav: Boolean(document.querySelector('nav.sticky')),
      frNames: document.querySelectorAll('[data-fr-name]').length,
      bodyFont: getComputedStyle(document.body).fontFamily,
      headingFont: h2 ? getComputedStyle(h2).fontFamily : '',
    };
  });
  const problems = [];
  if (response?.status() !== 200) problems.push(`status ${response?.status()}`);
  if (r.scrollWidth > r.clientWidth) problems.push(`horizontal scroll ${r.scrollWidth}>${r.clientWidth}`);
  if (r.pills.length !== 5) problems.push(`pills ${r.pills.length}`);
  if (r.pills.some((p) => p.w < 44 || p.h < 44)) problems.push('pill under 44px');
  const current = r.pills.filter((p) => p.current === 'page');
  if (current.length !== 1 || !current[0].href.endsWith(`/${locale}/menu`)) problems.push('current pill');
  if (!r.stickyNav) problems.push('no sticky quick-nav');
  if (locale === 'zh' && (r.lang !== 'zh-Hans' || !r.bodyFont.includes('PingFang SC') || !r.headingFont.includes('PingFang SC'))) problems.push(`zh lang/font: ${r.lang} | ${r.bodyFont} | ${r.headingFont}`);
  console.log(problems.length ? 'FAIL' : 'OK  ', locale, JSON.stringify({ ...r, pills: r.pills.map((p) => `${p.w}x${p.h}${p.current ? '*' : ''}`) }), problems.join('; '));
  if (problems.length) failures += 1;
  if (locale === 'zh') {
    await page.screenshot({ path: `${outDir}/zh-menu-375-top.png` });
    await page.getByText('价格已含服务费').first().scrollIntoViewIfNeeded();
    await page.screenshot({ path: `${outDir}/zh-menu-375-notes.png` });
  }
  await page.close();
}
await browser.close();
console.log(failures ? `MOBILE FAILED ${failures}` : 'MOBILE OK');
process.exit(failures ? 1 : 0);
```
