       import imageUrlBuilder from '@sanity/image-url';
       import type { SanityImageSource } from '@sanity/image-url/lib/types/types';
       import { client } from './client';

       const builder = imageUrlBuilder(client);

       export function urlFor(source: SanityImageSource) {
         return builder.image(source);

    3. Create `src/lib/localized.ts` — locale-aware field extractor for Sanity localeString/localeText objects:
       ```typescript
       export function localized(
         field: { fr: string; en?: string; de?: string } | null | undefined,
         locale: string,
         fallback = ''
       ): string {
         if (!field) return fallback;
         return field[locale as keyof typeof field] ?? field.fr ?? fallback;
       }
       ```

    4. Create `src/components/ui/FadeInUp.tsx` — reusable scroll animation wrapper using `m.div` from `motion/react` (NOT motion/react-m per Phase 1 decision — `m` is imported from `motion/react`):
       ```typescript
       'use client';

       import { m } from 'motion/react';

       interface FadeInUpProps {
         children: React.ReactNode;
         delay?: number;
         className?: string;
       }

       export function FadeInUp({ children, delay = 0, className }: FadeInUpProps) {
         return (
           <m.div
             initial={{ opacity: 0, y: 24 }}
             whileInView={{ opacity: 1, y: 0 }}
             viewport={{ once: true, margin: '-80px' }}
             transition={{ duration: 0.6, ease: 'easeOut', delay }}
             className={className}
           >
             {children}
           </m.div>
         );
       }
       ```
       IMPORTANT: Phase 1 decided `import { m } from 'motion/react'` is the correct import, NOT `import * as m from 'motion/react-m'`. See 01-03-SUMMARY.md decision. If whileInView animations fail silently during testing, implement fallback using `useInView` hook from `motion/react` (see 02-RESEARCH.md Pitfall 3).

         type: 'document',
