1. **src/app/globals.css** — Replace the minimal `@import "tailwindcss"` from Plan 01 with full UMAI design tokens (per research Pattern 4):
   ```css
   @import "tailwindcss";

   @theme {
     /* UMAI Colors */
     --color-umai-bg: #F5F0E8;
     --color-umai-bg-alt: #EDE8DC;
     --color-umai-text: #1A1A1A;
     --color-umai-text-muted: #6B6B6B;
     --color-umai-accent: #77967A;
     --color-umai-accent-hover: #657D67;
     --color-umai-line: #D4CFC5;
     --color-umai-line-dotted: #C5BFB3;
     --color-umai-black: #0A0A0A;
     --color-umai-white: #FFFFFF;

     /* Fonts — CSS variables from next/font, linked via @theme inline */
     --font-display: var(--font-dm-serif-display);
     --font-body: var(--font-outfit);
     --font-jp: var(--font-noto-sans-jp);

     /* Layout */
     --max-width-content: 1200px;
     --spacing-section: 120px;
     --spacing-section-mobile: 80px;
   }

   @theme inline {
     --font-dm-serif-display: ;
     --font-outfit: ;
     --font-noto-sans-jp: ;
   }
   ```
   The `@theme inline` block tells Tailwind these CSS variables exist but are set elsewhere (by next/font at runtime). This avoids Tailwind purging `font-display`, `font-body`, `font-jp` utilities.

   Also add base styles below the @theme blocks:
   ```css
   @layer base {
     body {
       @apply bg-umai-bg text-umai-text font-body antialiased;
     }
     h1, h2 {
       @apply font-display;
     }
   }
   ```

4. **src/components/layout/MotionProvider.tsx** — LazyMotion wrapper (per research Pattern 10):
   ```tsx
   'use client'

   import { LazyMotion, domAnimation } from 'motion/react';

   export function MotionProvider({ children }: { children: React.ReactNode }) {
     return <LazyMotion features={domAnimation}>{children}</LazyMotion>;
   }
   ```
   This reduces animation bundle from ~34kb to ~4.6kb. All animated components MUST use `m.div` from `motion/react-m`, NOT `motion.div`.
