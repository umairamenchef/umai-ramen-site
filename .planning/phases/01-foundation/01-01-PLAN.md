     let locale = await requestLocale;
     if (!hasLocale(routing.locales, locale)) {

6. **src/app/[locale]/layout.tsx** — Locale layout with ALL required next-intl patterns for static rendering:
   ```tsx
   import { NextIntlClientProvider, hasLocale } from 'next-intl';
   import { setRequestLocale, getMessages } from 'next-intl/server';
   import { notFound } from 'next/navigation';
   import { routing } from '@/i18n/routing';

   export const revalidate = 60; // FOUND-04: ISR 60s baseline

   export function generateStaticParams() {
     return routing.locales.map((locale) => ({ locale }));
   }

   type Props = {
     children: React.ReactNode;
     params: Promise<{ locale: string }>;
   };

   export default async function LocaleLayout({ children, params }: Props) {
     const { locale } = await params;
     if (!hasLocale(routing.locales, locale)) {
       notFound();
     }
     setRequestLocale(locale);
     const messages = await getMessages();

     return (
       <html lang={locale}>
         <body>
           <NextIntlClientProvider messages={messages}>
             {children}
           </NextIntlClientProvider>
         </body>
       </html>
     );
   }
   ```
   CRITICAL: `await params` (async, not sync — Next.js 16 breaking change). `setRequestLocale` BEFORE any next-intl call. `generateStaticParams` returns all locales for static rendering.
